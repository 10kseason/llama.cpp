set(TARGET llama-s24-bench)
add_executable(${TARGET} s24-bench.cpp)
target_include_directories(${TARGET} PRIVATE "${PROJECT_SOURCE_DIR}/ggml/src" "${PROJECT_SOURCE_DIR}/vendor")
target_link_libraries(${TARGET} PRIVATE ggml ggml-vulkan ${CMAKE_THREAD_LIBS_INIT})
target_compile_features(${TARGET} PRIVATE cxx_std_17)
if (MINGW)
    target_link_options(${TARGET} PRIVATE -municode)
endif()
install(TARGETS ${TARGET} RUNTIME)

if (LLAMA_BUILD_TESTS)
    add_test(NAME s24-vulkan-self-test COMMAND $<TARGET_FILE:${TARGET}> --self-test)
    set_tests_properties(s24-vulkan-self-test PROPERTIES LABELS "gpu;s24" TIMEOUT 180)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if (Python3_Interpreter_FOUND)
        add_test(NAME s24-vulkan-rejections COMMAND "${Python3_EXECUTABLE}" -B
            "${CMAKE_CURRENT_SOURCE_DIR}/test-rejections.py" $<TARGET_FILE:${TARGET}>)
        set_tests_properties(s24-vulkan-rejections PROPERTIES LABELS "gpu;s24" TIMEOUT 180)
    endif()
endif()
